-- ════════════════════════════════════════════════════════════════
--  FLY
-- ════════════════════════════════════════════════════════════════
local function setFly(v)
    S.Fly = v
    if S.FlyConn then S.FlyConn:Disconnect() S.FlyConn = nil end

    local hum = getHum()
    local hrp = getHRP()
    if not hum or not hrp then return end

    if v then
        hum.PlatformStand = true
        local bg = Instance.new("BodyGyro")
        bg.MaxTorque = Vector3.new(1e9,1e9,1e9) bg.P = 1e4 bg.D = 1e3 bg.Parent = hrp

        local bv = Instance.new("BodyVelocity")
        bv.MaxForce = Vector3.new(1e9,1e9,1e9) bv.Velocity = Vector3.zero bv.Parent = hrp

        S.FlyConn = RunService.Heartbeat:Connect(function()
            local hrp2 = getHRP() if not hrp2 then return end
            if not bg.Parent then return end
            local spd = S.FlySpeed
            local dir = Vector3.zero
            local cf  = Camera.CFrame
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + cf.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - cf.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - cf.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + cf.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0,1,0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0,1,0) end
            bv.Velocity = dir.Magnitude > 0 and dir.Unit * spd or Vector3.zero
            bg.CFrame = Camera.CFrame
        end)
    else
        -- cleanup
        if hrp then
            for _, c in ipairs(hrp:GetChildren()) do
                if c:IsA("BodyGyro") or c:IsA("BodyVelocity") then c:Destroy() end
            end
        end
        if hum then hum.PlatformStand = false end
    end
end

-- ════════════════════════════════════════════════════════════════
--  SILENT AIM (Survivor / Killer)
-- ════════════════════════════════════════════════════════════════
local function getBestTarget(targetType)
    local best, bestDist = nil, S.AimRadius
    local center = getScreenCenter()

    for _, p in ipairs(Players:GetPlayers()) do
        if p == LP then continue end
        local char = p.Character
        if not char then continue end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hrp or not hum or hum.Health <= 0 then continue end

        -- filter by target type (check naming or group)
        local isKiller = char:FindFirstChild("KillerConfig") ~= nil
            or char:FindFirstChild("KillerMorph") ~= nil
            or char.Name:lower():find("killer")
        local isSurvivor = not isKiller

        local ok = false
        if targetType == "Killer" and isKiller then ok = true
        elseif targetType == "Survivor" and isSurvivor then ok = true
        elseif targetType == "All" then ok = true end

        if not ok then continue end

        local vp, inView = worldToViewport(hrp.Position)
        if not inView then continue end
        local dist = (vp - center).Magnitude
        if dist < bestDist then
            bestDist = dist
            best = hrp
        end
    end
    return best
end

-- ════════════════════════════════════════════════════════════════
local function findNearestGen()
    local hrp = getHRP() if not hrp then return nil end
    local best, bestD = nil, math.huge
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj.Name == "Generator" or obj.Name == "Gen" or obj.Name == "GeneratorPoint" then
            if obj:IsA("BasePart") or obj:IsA("Model") then
                local pos = obj:IsA("Model") and obj:GetModelCFrame().Position
                    or obj.Position
                local d = (pos - hrp.Position).Magnitude
                if d < bestD then bestD = d best = obj end
            end
        end
    end
    return best
end

local function fireRepairEvent()
    safe(function()
        local rs = ReplicatedStorage
        local repairEvent = rs:FindFirstChild("RepairEvent", true)
            or rs:FindFirstChild("GeneratorRepair", true)
            or rs:FindFirstChild("Repair", true)
        if repairEvent then
            if repairEvent:IsA("RemoteEvent") then repairEvent:FireServer()
            elseif repairEvent:IsA("BindableEvent") then repairEvent:Fire() end
        end
        -- Also try skillcheck auto-success
        local skillOk = rs:FindFirstChild("SkillCheckResultEvent", true)
            or rs:FindFirstChild("SkillcheckValidated", true)
        if skillOk then
            if skillOk:IsA("RemoteEvent") then skillOk:FireServer(true)
            elseif skillOk:IsA("BindableEvent") then skillOk:Fire(true) end
        end
    end)
end

local function findKillerHRP()
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LP then continue end
        local char = p.Character if not char then continue end
        if char:FindFirstChild("KillerConfig") or char:FindFirstChild("KillerMorph") then
            return char:FindFirstChild("HumanoidRootPart"), p
        end
    end
    -- fallback: look for KillerSpawn model
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj.Name == "KillerSpawn" and obj:IsA("BasePart") then
            return obj
        end
    end
    return nil
end

local function killerFacingSurvivor(killerHRP)
    local myHRP = getHRP() if not myHRP or not killerHRP then return true end
    local toSurvivor = (myHRP.Position - killerHRP.Position).Unit
    local killerLook = killerHRP.CFrame.LookVector
    local dot = killerLook:Dot(toSurvivor)
    local angle = math.deg(math.acos(math.clamp(dot, -1, 1)))
    return angle <= (S.FaceKillerAngle / 2)
end

local function setAutoParry(v)
    S.AutoParry = v
    if S.ParryConn then S.ParryConn:Disconnect() S.ParryConn = nil end
    if not v then return end
    S.ParryConn = RunService.Heartbeat:Connect(function()
        if not S.AutoParry then return end
        local myHRP = getHRP() if not myHRP then return end
        local killerHRP = findKillerHRP()
        if not killerHRP then return end
        local dist = (killerHRP.Position - myHRP.Position).Magnitude
        if dist <= S.ParryRange then
            if not killerFacingSurvivor(killerHRP) then return end
            task.wait(S.ParryDelay)
            doParry()
        end
    end)
end

-- ════════════════════════════════════════════════════════════════
--  KILLER: SNAP LINE (draw beam to target)
-- ════════════════════════════════════════════════════════════════
local snapLineGui = nil
local function buildSnapLine()
    local pgui = LP:FindFirstChild("PlayerGui")
    if not pgui then return end
    local old = pgui:FindFirstChild("RYZ_SnapLine")
    if old then old:Destroy() end
    if not S.SnapLine then return end

    local sg = Instance.new("ScreenGui")
    sg.Name = "RYZ_SnapLine" sg.ResetOnSpawn=false sg.ZIndexBehavior=Enum.ZIndexBehavior.Sibling
    sg.DisplayOrder=9998 sg.Parent=pgui
    snapLineGui = sg

    local frame = Instance.new("Frame")
    frame.Size=UDim2.new(1,0,1,0) frame.BackgroundTransparency=1 frame.Parent=sg

    RunService.RenderStepped:Connect(function()
        if not S.SnapLine then frame:ClearAllChildren() return end
        frame:ClearAllChildren()
        local target = getBestTarget("Survivor")
        if not target then return end
        local myHRP = getHRP() if not myHRP then return end

        local vpA, inA = worldToViewport(myHRP.Position)
        local vpB, inB = worldToViewport(target.Position)
        if not inA or not inB then return end

        local dist = (vpB - vpA).Magnitude
        local angle = math.atan2(vpB.Y - vpA.Y, vpB.X - vpA.X)
        local mid = (vpA + vpB) / 2

        local line = Instance.new("Frame")
        line.Size = UDim2.fromOffset(dist, 2)
        line.Position = UDim2.fromOffset(mid.X - dist/2, mid.Y - 1)
        line.AnchorPoint = Vector2.new(0.5, 0.5)
        line.BackgroundColor3 = Color3.fromRGB(255, 60, 60)
        line.BackgroundTransparency = 0.3
        line.BorderSizePixel = 0
        line.Rotation = math.deg(angle)
        line.ZIndex = 5
        local uc = Instance.new("UICorner") uc.CornerRadius=UDim.new(1,0) uc.Parent=line
        line.Parent = frame
    end)
end

-- ════════════════════════════════════════════════════════════════
--  KILLER: AIM PREDICT (lead target)
-- ════════════════════════════════════════════════════════════════
local function getPredictedPos(hrp, projectileSpeed)
    local vel = hrp.AssemblyLinearVelocity
    local myHRP = getHRP() if not myHRP then return hrp.Position end
    local dist = (hrp.Position - myHRP.Position).Magnitude
    local travelTime = dist / (projectileSpeed or 80)
    return hrp.Position + vel * travelTime
end

-- ════════════════════════════════════════════════════════════════
--  ESP SYSTEM (Highlight-based)
-- ════════════════════════════════════════════════════════════════
local ESPData = {}

local function removeESP(model)
    if ESPData[model] then
        safe(function()
            if ESPData[model].highlight and ESPData[model].highlight.Parent then
                ESPData[model].highlight:Destroy()
            end
            if ESPData[model].billGui and ESPData[model].billboard.Parent then
                ESPData[model].billboard:Destroy()
            end
        end)
        ESPData[model] = nil
    end
end

local function clearAllESP()
    for model, _ in pairs(ESPData) do removeESP(model) end
    ESPData = {}
end

local function makeHighlight(model, fillColor, outlineColor, fillTrans, outlineTrans)
    if ESPData[model] and ESPData[model].highlight then
        safe(function() ESPData[model].highlight:Destroy() end)
    end
    local hl = Instance.new("SelectionBox")
    hl.Adornee = model
    hl.Color3 = outlineColor or Color3.fromRGB(255,255,255)
    hl.SurfaceColor3 = fillColor or Color3.fromRGB(255,255,255)
    hl.SurfaceTransparency = fillTrans or 0.7
    hl.LineThickness = 0.04
    hl.Parent = Workspace
    return hl
end

local function makeBillboard(part, title, color)
    local bg = Instance.new("BillboardGui")
    bg.Adornee = part
    bg.AlwaysOnTop = true
    bg.Size = UDim2.fromOffset(200, 60)
    bg.StudsOffsetWorldSpace = Vector3.new(0, 3.5, 0)
    bg.Parent = part

    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(1, 0, 1, 0)
    lbl.BackgroundTransparency = 1
    lbl.Text = title
    lbl.TextColor3 = color or Color3.fromRGB(255, 255, 255)
    lbl.TextSize = 13
    lbl.Font = Enum.Font.GothamBold
    lbl.TextStrokeTransparency = 0
    lbl.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    lbl.Parent = bg

    return bg, lbl
end

local function updateESPLabel(model, lbl, extraInfo)
    if not lbl or not lbl.Parent then return end
    local hrp = model:FindFirstChild("HumanoidRootPart")
    local myHRP = getHRP()
    local dist = (hrp and myHRP) and math.floor((hrp.Position - myHRP.Position).Magnitude) or "?"
    local hum = model:FindFirstChildOfClass("Humanoid")
    local hp = hum and math.floor(hum.Health) or "?"
    lbl.Text = string.format("[%s]\nHP:%s  DIST:%sm%s", model.Name, hp, dist, extraInfo or "")
end

-- ── SURVIVOR ESP ─────────────────────────────────────────────
local function refreshSurvivorESP()
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LP then continue end
        local char = p.Character
        if not char then continue end
        local isKiller = char:FindFirstChild("KillerConfig") ~= nil

        if not isKiller and S.ESPSurvivor then
            if not ESPData[char] then
                local hl = makeHighlight(char, Color3.fromRGB(0,200,255), Color3.fromRGB(0,200,255), 0.75, 0)
                local hrp = char:FindFirstChild("HumanoidRootPart")
                local bg, lbl = makeBillboard(hrp or char.PrimaryPart or char:FindFirstChild("Head") or char, "", Color3.fromRGB(0,200,255))
                ESPData[char] = {highlight=hl, billboard=bg, label=lbl, type="Survivor"}
            end
        elseif not isKiller and not S.ESPSurvivor then
            if ESPData[char] and ESPData[char].type == "Survivor" then removeESP(char) end
        end
    end
end

-- ── KILLER ESP ───────────────────────────────────────────────
local function getKillerItem(char)
    for _, c in ipairs(char:GetChildren()) do
        if c:IsA("Tool") then return c.Name end
    end
    return "None"
end

local function refreshKillerESP()
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LP then continue end
        local char = p.Character
        if not char then continue end
        local isKiller = char:FindFirstChild("KillerConfig") ~= nil
            or char:FindFirstChild("KillerMorph") ~= nil

        if isKiller and S.ESPKiller then
            if not ESPData[char] then
                local hl = makeHighlight(char, Color3.fromRGB(255,50,50), Color3.fromRGB(255,50,50), 0.65, 0)
                local hrp = char:FindFirstChild("HumanoidRootPart")
                local bg, lbl = makeBillboard(hrp or char:FindFirstChild("Head") or char, "", Color3.fromRGB(255,80,80))
                ESPData[char] = {highlight=hl, billboard=bg, label=lbl, type="Killer"}
            end
        elseif isKiller and not S.ESPKiller then
            if ESPData[char] and ESPData[char].type == "Killer" then removeESP(char) end
        end
    end
end

-- ── GENERATOR ESP ────────────────────────────────────────────
local function refreshGenESP()
    if not S.ESPGenerator then
        for model, data in pairs(ESPData) do
            if data.type == "Generator" then removeESP(model) end
        end
        return
    end
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if (obj.Name == "Generator" or obj.Name == "Gen") and obj:IsA("Model") then
            if not ESPData[obj] then
                local hl = makeHighlight(obj, Color3.fromRGB(255,200,0), Color3.fromRGB(255,200,0), 0.7, 0)
                local pp = obj.PrimaryPart or obj:FindFirstChildOfClass("BasePart")
                local bg, lbl = makeBillboard(pp, "", Color3.fromRGB(255,200,0))
                ESPData[obj] = {highlight=hl, billboard=bg, label=lbl, type="Generator"}
            end
        end
    end
end

-- ── HOOK ESP ─────────────────────────────────────────────────
local function refreshHookESP()
    if not S.ESPHook then
        for model, data in pairs(ESPData) do
            if data.type == "Hook" then removeESP(model) end
        end
        return
    end
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj.Name == "Hook" or obj.Name == "HookPoint" then
            local target = obj:IsA("BasePart") and obj or (obj:IsA("Model") and obj:FindFirstChildOfClass("BasePart"))

                hl.SurfaceColor3=Color3.fromRGB(180,90,255) hl.SurfaceTransparency=0.75
                hl.LineThickness=0.04 hl.Parent=Workspace
                local bg, lbl = makeBillboard(target, "🪝 HOOK", Color3.fromRGB(200,120,255))
                ESPData[obj] = {highlight=hl, billboard=bg, label=lbl, type="Hook"}
            end
        end
    end
end

-- ── KILLER NOTIF (! above survivor near killer) ──────────────
local notifBills = {}
local function refreshKillerNotif()
    -- Clear old
    for char, bg in pairs(notifBills) do
        if bg and bg.Parent then bg:Destroy() end
    end
    notifBills = {}
    if not S.KillerNotif then return end
end

local function updateKillerNotif()
    if not S.KillerNotif then return end
    local killerHRP = findKillerHRP()
    if not killerHRP then return end

    for _, p in ipairs(Players:GetPlayers()) do
        if p == LP then continue end
        local char = p.Character if not char then continue end
        local hrp = char:FindFirstChild("HumanoidRootPart") if not hrp then continue end
        local isKiller = char:FindFirstChild("KillerConfig") ~= nil
        if isKiller then continue end

        local dist = (hrp.Position - killerHRP.Position).Magnitude
        local head = char:FindFirstChild("Head") or hrp

        if dist < 25 then
            if not notifBills[char] then
                local bg = Instance.new("BillboardGui")
                bg.Adornee=head bg.AlwaysOnTop=true
                bg.Size=UDim2.fromOffset(50,50)
                bg.StudsOffsetWorldSpace=Vector3.new(0,5,0)
                bg.Parent=head
                local lbl=Instance.new("TextLabel")
                lbl.Size=UDim2.new(1,0,1,0)
                lbl.BackgroundTransparency=1 lbl.Text="⚠"
                lbl.TextColor3=Color3.fromRGB(255,80,0)
                lbl.TextSize=30 lbl.Font=Enum.Font.GothamBold
                lbl.TextStrokeTransparency=0 lbl.Parent=bg
                notifBills[char] = bg
            end
        else
            if notifBills[char] then
                safe(function() notifBills[char]:Destroy() end)
                notifBills[char] = nil
            end
        end
    end
end

local CROSSHAIRS = {
    ["Dot"]     = { shape="dot",   size=6,  color=Color3.fromRGB(255,255,255) },
    ["Cross"]   = { shape="cross", size=16, color=Color3.fromRGB(255,255,255) },
    ["Circle"]  = { shape="circle",size=20, color=Color3.fromRGB(255,255,255) },
    ["T-Shape"] = { shape="tshape",size=16, color=Color3.fromRGB(255,80,80)   },
    ["Plus"]    = { shape="plus",  size=16, color=Color3.fromRGB(0,220,100)   },
    ["None"]    = { shape="none",  size=0,  color=Color3.fromRGB(0,0,0)       },
}

local function buildCrosshair(style)
    local pgui = LP:FindFirstChild("PlayerGui") if not pgui then return end
    local old = pgui:FindFirstChild("RYZ_Crosshair") if old then old:Destroy() end
    S.CrosshairGui = nil
    if not style or style == "None" then return end

    local cfg = CROSSHAIRS[style]
    if not cfg then return end

    local sg = Instance.new("ScreenGui")
    sg.Name="RYZ_Crosshair" sg.ResetOnSpawn=false sg.DisplayOrder=9990 sg.IgnoreGuiInset=true sg.Parent=pgui
    S.CrosshairGui = sg

    local vp = Camera.ViewportSize
    local cx, cy = vp.X/2, vp.Y/2
    local s = cfg.size
    local col = cfg.color

    local function rect(x,y,w,h)
        local f=Instance.new("Frame")
        f.Position=UDim2.fromOffset(x,y) f.Size=UDim2.fromOffset(w,h)
        f.BackgroundColor3=col f.BorderSizePixel=0
        f.ZIndex=10 f.Parent=sg
        return f
    end

    if cfg.shape == "dot" then
        local r = rect(cx-s/2, cy-s/2, s, s)
        local uc=Instance.new("UICorner") uc.CornerRadius=UDim.new(1,0) uc.Parent=r
    elseif cfg.shape == "cross" or cfg.shape == "plus" then
        rect(cx-s/2, cy-1, s, 2)   -- horizontal
        rect(cx-1, cy-s/2, 2, s)   -- vertical
    elseif cfg.shape == "circle" then
        local f=Instance.new("Frame")
        f.Position=UDim2.fromOffset(cx-s/2, cy-s/2) f.Size=UDim2.fromOffset(s,s)
        f.BackgroundTransparency=1 f.BorderSizePixel=0 f.ZIndex=10 f.Parent=sg
        local ui=Instance.new("UICorner") ui.CornerRadius=UDim.new(1,0) ui.Parent=f
        local st=Instance.new("UIStroke") st.Color=col st.Thickness=1.5 st.Parent=f
    elseif cfg.shape == "tshape" then
        rect(cx-s/2, cy, s, 2)      -- horizontal bottom
        rect(cx-1,   cy-s/2, 2, s/2) -- vertical top only
    end
end

local function setNoTexture(v)
    S.NoTexture = v
    safe(function()
        if v then
            for _, obj in ipairs(Workspace:GetDescendants()) do
                if obj:IsA("Texture") or obj:IsA("Decal") then
                    if not S.OrigTextures[obj] then
                        S.OrigTextures[obj] = obj.Transparency
                    end
                    obj.Transparency = 1
                end
            end
        else
            for obj, orig in pairs(S.OrigTextures) do
                safe(function() if obj and obj.Parent then obj.Transparency = orig end end)
            end
            S.OrigTextures = {}
        end
    end)
end

local CONFIG_FOLDER = "RYZ_VD_Configs"
local function ensureConfigFolder()
    safe(function() if not isfolder(CONFIG_FOLDER) then makefolder(CONFIG_FOLDER) end end)
end
local function getConfigList()
    local list = {}
    safe(function()
        ensureConfigFolder()
        for _, f in ipairs(listfiles(CONFIG_FOLDER)) do
            local n = f:match("([^/\\]+)$") or f
            if n:sub(-5)==".json" then table.insert(list, n:sub(1,-6)) end
        end
    end)
    return #list>0 and list or {"(No Configs)"}
end

local function serializeConfig()
    return HttpService:JSONEncode({
        NoClip=S.NoClip, Fly=S.Fly, FlySpeed=S.FlySpeed,
        SpeedHack=S.SpeedHack, SpeedValue=S.SpeedValue,
        InfJump=S.InfJump, AntiAFK=S.AntiAFK,
        AutoParry=S.AutoParry, ParryRange=S.ParryRange,
        FaceKillerAngle=S.FaceKillerAngle, ParryDelay=S.ParryDelay,
        SilentAim=S.SilentAim, AimRadius=S.AimRadius,
        AimSmooth=S.AimSmooth, SilentLaser=S.SilentLaser,
        AimTarget=S.AimTarget, AutoGen=S.AutoGen,
        KillerSilentAim=S.KillerSilentAim, SpearGravity=S.SpearGravity,
        SnapLine=S.SnapLine, AimPredict=S.AimPredict,
        AntiStun=S.AntiStun, AntiFlashlight=S.AntiFlashlight,
        ESPSurvivor=S.ESPSurvivor, ESPKiller=S.ESPKiller,
        ESPGenerator=S.ESPGenerator, ESPHook=S.ESPHook,
        ESPPallet=S.ESPPallet, KillerNotif=S.KillerNotif,
        CamFOV=S.CamFOV, Crosshair=S.Crosshair,
        NoTexture=S.NoTexture, FullBright=S.FullBright, NoFog=S.NoFog,
    })
end

local function saveConfig(name)
    if not name or name=="" then return false end
    ensureConfigFolder()
    safe(function()
        writefile(CONFIG_FOLDER.."/"..name..".json", serializeConfig())
    end)
    return true
end

local function loadConfig(name)
    local raw
    safe(function() raw = readfile(CONFIG_FOLDER.."/"..name..".json") end)
    if not raw then return false end
    local ok, data = pcall(function() return HttpService:JSONDecode(raw) end)
    if not ok or type(data) ~= "table" then return false end

    for k,v in pairs(data) do
        if S[k] ~= nil then S[k] = v end
    end

    -- Re-apply all settings
    setNoClip(S.NoClip)
    setFly(false) if S.Fly then setFly(true) end
    applySpeed()
    setAutoGen(S.AutoGen)
    setAutoParry(S.AutoParry)
    buildSnapLine()
    setFOV(S.CamFOV)
    buildCrosshair(S.Crosshair)
    setNoTexture(S.NoTexture)
    setFullBright(S.FullBright)
    setNoFog(S.NoFog)
    return true
end

local function deleteConfig(name)
    safe(function() delfile(CONFIG_FOLDER.."/"..name..".json") end)
end

--  TAB 6 — CONFIG
-- ════════════════════════════════════════════════════════════════
local TCfg = Window:Tab({ Title="Config", Icon="folder" })

local configName = ""
local selConfig = ""
local configDD

local function refreshConfigDD()
    local list = getConfigList()
    selConfig = list[1]
    pcall(function() configDD:Refresh(list) end)
end

configDD = TCfg:Dropdown({
    Title="Config List", Icon="list",
    Values=getConfigList(), Value=getConfigList()[1],
    Callback=function(v) selConfig=v end
})

TCfg:Input({
    Title="Nama Config", Icon="file-text",
    Placeholder="contoh: MySetting",
    Callback=function(v) configName=v end
})

TCfg:Button({
    Title="Save Config", Icon="save", Color=Color3.fromHex("#22c55e"),
    Callback=function()
        local name = configName~="" and configName or ("config_"..os.time())
        if saveConfig(name) then
            notify("Config","Tersimpan: "..name,"save",3)
            refreshConfigDD()
        else
            notify("Config","Gagal simpan!","alert-circle",3)
        end
    end
})
TCfg:Button({
    Title="Load Config", Icon="folder-open", Color=Color3.fromHex("#3b82f6"),
    Callback=function()
        if selConfig=="" or selConfig=="(No Configs)" then
            notify("Config","Pilih config dulu!","alert-circle",2) return
        end
        if loadConfig(selConfig) then
            notify("Config","Loaded: "..selConfig,"folder-open",3)
        else
            notify("Config","Gagal load!","alert-circle",3)
        end
    end
})
TCfg:Button({
    Title="Delete Config", Icon="trash-2", Color=Color3.fromHex("#ef4444"),
    Callback=function()
        if selConfig=="" or selConfig=="(No Configs)" then
            notify("Config","Pilih config dulu!","alert-circle",2) return
        end
        deleteConfig(selConfig)
        notify("Config","Dihapus: "..selConfig,"trash-2",3)
        refreshConfigDD()
    end
})

TCfg:Divider()
