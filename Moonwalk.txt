local function loadConfigs()

local function isKiller(p)
    local t = p.Team
    return t and t.Name == "Killer"
end
local function isSurvivor(p)
    local t = p.Team
    return t and t.Name == "Survivor"
end
local function isZombie(p)
    -- zombie flagged by attribute atau team name
    local c = p.Character
    if not c then return false end
    return c:GetAttribute("IsZombie") == true
end

local function getKillers()
    local t={}
    for _,p in ipairs(Players:GetPlayers()) do
        if p~=LP and isKiller(p) and p.Character then table.insert(t,p) end
    end
    return t
end
local function getSurvivors()
    local t={}
    for _,p in ipairs(Players:GetPlayers()) do
        if p~=LP and isSurvivor(p) and p.Character then table.insert(t,p) end
    end
    return t
end

-- Closest player matching target type (for silent aim)
local function getClosestToMouse(targetType)
    local best, bestDist = nil, math.huge
    local mousePos = UIS:GetMouseLocation()
    local screenCenter = Vector2.new(
        Cam.ViewportSize.X / 2,
        Cam.ViewportSize.Y / 2
    )
    local aimPos = Vector2.new(mousePos.X, mousePos.Y)

    for _,p in ipairs(Players:GetPlayers()) do
        if p == LP then continue end
        local char = p.Character
        if not char then continue end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hrp or not hum or hum.Health <= 0 then continue end

        -- Filter by type
        local ok = false
        if targetType == "Killer"   and isKiller(p)   then ok = true end
        if targetType == "Survivor" and isSurvivor(p) then ok = true end
        if targetType == "Zombie"   and isZombie(p)   then ok = true end

        if ok then
            local screenPt, vis = Cam:WorldToViewportPoint(hrp.Position)
            if vis then
                local d = (Vector2.new(screenPt.X, screenPt.Y) - aimPos).Magnitude
                if d < S.aimRadius and d < bestDist then
                    bestDist = d
                    best = p
                end
            end
        end
    end
    return best
end

-- Direction from HRP to target (for aim prediction)
local function predictPos(hrp, tVel, dist)
    local speed = dist / 0.08   -- approx projectile speed
    return hrp.Position + tVel * (dist / speed)
end

local function playerList()

    for _,p in ipairs(Players:GetPlayers()) do
        if p~=LP then table.insert(t, p.Name) end
    end
    return #t>0 and t or {"(nobody)"}
end

local function flyStart()
    if S.flyOn then return end
    local _,hum,root = gc(); if not root then return end
    S.flyOn = true; if hum then hum.PlatformStand = true end
    S.flyBV = Instance.new("BodyVelocity")
    S.flyBV.MaxForce = Vector3.new(1e9,1e9,1e9)
    S.flyBV.Velocity  = Vector3.zero
    S.flyBV.Parent    = root
    S.flyBG = Instance.new("BodyGyro")
    S.flyBG.MaxTorque = Vector3.new(1e9,1e9,1e9)
    S.flyBG.D = 100; S.flyBG.Parent = root
    S.flyConn = RunService.Heartbeat:Connect(function()
        if not S.flyOn then S.flyConn:Disconnect(); return end
        local d = Vector3.zero; local spd = 55
        if UIS:IsKeyDown(Enum.KeyCode.W) then d = d + Cam.CFrame.LookVector end
        if UIS:IsKeyDown(Enum.KeyCode.S) then d = d - Cam.CFrame.LookVector end
        if UIS:IsKeyDown(Enum.KeyCode.A) then d = d - Cam.CFrame.RightVector end
        if UIS:IsKeyDown(Enum.KeyCode.D) then d = d + Cam.CFrame.RightVector end
        if UIS:IsKeyDown(Enum.KeyCode.Space)       then d = d + Vector3.new(0,1,0) end
        if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then d = d - Vector3.new(0,1,0) end
        if UIS:IsKeyDown(Enum.KeyCode.LeftShift)   then spd = spd * 2.5 end
        S.flyBV.Velocity = d.Magnitude > 0 and d.Unit * spd or Vector3.zero
        S.flyBG.CFrame   = Cam.CFrame
    end)
end
local function flyStop()
    S.flyOn = false
    if S.flyConn then S.flyConn:Disconnect(); S.flyConn=nil end
    if S.flyBV   then S.flyBV:Destroy(); S.flyBV=nil end
    if S.flyBG   then S.flyBG:Destroy(); S.flyBG=nil end
    local _,hum = gc(); if hum then hum.PlatformStand = false end
end

-- ══════════════════════════════════════════════════════════
-- [11]  AUTO PARRY
--   Deteksi killer dalam range + sudut → FireServer parry
-- ══════════════════════════════════════════════════════════
local function autoParryStart()
    if S.autoParryOn then return end; S.autoParryOn = true
    S.parryThread = task.spawn(function()
        local lastParry = 0
        while S.autoParryOn do
            task.wait(0.05)
            local _,_,root = gc(); if not root then continue end
            local killers = getKillers()
            if #killers == 0 then continue end

            for _, killer in ipairs(killers) do
                local kChar = killer.Character
                if not kChar then continue end
                local kHRP = kChar:FindFirstChild("HumanoidRootPart")
                if not kHRP then continue end

                local dist = (root.Position - kHRP.Position).Magnitude
                if dist > S.parryRange then continue end

                -- Face killer angle check
                local toKiller = (kHRP.Position - root.Position).Unit
                local rootLook = root.CFrame.LookVector
                local dot      = rootLook:Dot(toKiller)
                local angleDeg = math.deg(math.acos(math.clamp(dot, -1, 1)))

                if angleDeg > S.faceKillerAngle then continue end

                -- Cooldown + delay
                if tick() - lastParry < (S.parryDelay + 0.5) then continue end
                lastParry = tick()

                task.wait(S.parryDelay)
                pcall(function() R_ParryEvent:FireServer() end)
                S.status = "⚔️ Parried!"
            end
        end
        S.status = "Idle"
    end)
end
local function autoParryStop()
    S.autoParryOn = false
    if S.parryThread then task.cancel(S.parryThread); S.parryThread=nil end
end

--   Hook camera ray ke target terdekat dalam aimRadius

local origRaycast = workspace.Raycast
local silentAimConn

local function silentAimStart()
    if silentAimConn then return end
    -- Override Raycast untuk redirect ke target
    silentAimConn = RunService.RenderStepped:Connect(function()
        if not S.silentAimOn then return end
        local target = getClosestToMouse(S.silentAimTarget)
        if not target then S.silentAimTarget_locked = nil; return end
        local char = target.Character
        if not char then return end
        local hrp = char:FindFirstChild("HumanoidRootPart"); if not hrp then return end
        S.silentAimTarget_locked = hrp

        -- Smooth aim: lerp camera LookVector toward target
        local tPos = hrp.Position
        local camPos = Cam.CFrame.Position
        local dir = (tPos - camPos).Unit
        local smooth = math.clamp(S.aimSmooth / 100, 0.01, 1)

        local newLook = Cam.CFrame.LookVector:Lerp(dir, smooth)
        Cam.CFrame = CFrame.new(camPos, camPos + newLook)
    end)
end

local function silentAimStop()
    S.silentAimTarget_locked = nil
    if silentAimConn then silentAimConn:Disconnect(); silentAimConn=nil end
end

-- [14]  AUTO GENERATOR
--   Scan semua generator aktif → repair otomatis via RemoteEvent
-- ══════════════════════════════════════════════════════════
local function getActiveGenerators()
    local gens = {}
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj.Name == "Generator" or obj.Name == "generator" then
            local prog = obj:GetAttribute("RepairProgress") or 0
            if prog < 100 then
                table.insert(gens, {obj=obj, prog=prog})
            end
        end
    end
    -- Sort by lowest progress first
    table.sort(gens, function(a,b) return a.prog < b.prog end)
    return gens
end

local function tpTo(pos)
    local _,_,root = gc()
    if root then root.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0)) end
    task.wait(0.2)
end

local function autoGenStart()
    if S.autoGenOn then return end; S.autoGenOn = true
    S.autoGenThread = task.spawn(function()
        while S.autoGenOn do
            local gens = getActiveGenerators()
            if #gens == 0 then
                S.status = "All gens done!"
                task.wait(5); continue
            end
            local target = gens[1]
            local genHRP = target.obj:FindFirstChild("HumanoidRootPart")
                        or target.obj:FindFirstChild("PrimaryPart")
                        or target.obj
            local pos = genHRP:IsA("BasePart") and genHRP.Position or target.obj.Position
            tpTo(pos)
            S.status = "🔧 Repairing gen (" .. math.floor(target.prog) .. "%)"
            -- Fire repair event
            pcall(function() R_RepairEvent:FireServer(target.obj, true) end)
            pcall(function() R_RepairAnim:FireServer(target.obj) end)
            task.wait(0.3)

            -- Repair loop until 100
            local timeout = 0
            while S.autoGenOn and timeout < 60 do
                local prog = target.obj:GetAttribute("RepairProgress") or 0
                if prog >= 100 then break end
                pcall(function() R_RepairEvent:FireServer(target.obj, true) end)
                S.status = "🔧 Gen " .. math.floor(prog) .. "%"
                timeout = timeout + 0.4
                task.wait(0.4)
            end
            -- Stop repair
            pcall(function() R_RepairEvent:FireServer(target.obj, false) end)
            task.wait(1)
        end
        S.status = "Idle"
    end)
end
local function autoGenStop()
    S.autoGenOn = false
    if S.autoGenThread then task.cancel(S.autoGenThread); S.autoGenThread=nil end
    pcall(function()
        local gens = getActiveGenerators()
        if gens[1] then R_RepairEvent:FireServer(gens[1].obj, false) end
    end)
end

local function spearGravityStart()
    S.spearGravityOn = true
    -- Hook: when Spearthrow fires, override gravity on spear projectile
    -- Listen to workspace for new spear parts
    workspace.DescendantAdded:Connect(function(d)
        if not S.spearGravityOn then return end
        if d.Name and (d.Name:lower():find("spear") or d.Name:lower():find("projectile")) then
            if d:IsA("BasePart") then
                pcall(function()
                    d.CustomPhysicalProperties = PhysicalProperties.new(
                        0.7, 0.3, 0.5, 0, 0)
                    -- Kill gravity via BodyForce
                    local bf = Instance.new("BodyForce")
                    bf.Force = Vector3.new(0, d:GetMass() * workspace.Gravity, 0)
                    bf.Parent = d
                end)
            end
        end
    end)
end

local function getStatusStr(char)
    local hum = char:FindFirstChildOfClass("Humanoid"); if not hum then return "?" end
    if hum.Health <= 0 then return "<font color='#FF0000'>Dead</font>" end
    local isHooked   = char:GetAttribute("IsHooked")   or false
    local isCarried  = char:GetAttribute("IsCarried")  or false
    if isHooked  then return "<font color='#FF8C00'>Hooked</font>" end
    if isCarried then return "<font color='#FFA500'>Carried</font>" end
    local hp = math.floor(hum.Health / hum.MaxHealth * 100)
    return "<font color='#00FF80'>HP:"..hp.."%</font>"
end

local crosshairObjs = {}

local function clearCrosshair()
    for _,d in pairs(crosshairObjs) do pcall(function() d:Remove() end) end
    crosshairObjs = {}
end

local CROSSHAIRS = {"None","Dot","Cross","Circle","Diamond","T-Shape","Sniper"}

local function drawCrosshair(ctype)
    clearCrosshair()
    if ctype == "None" then return end
    local cx = Cam.ViewportSize.X / 2
    local cy = Cam.ViewportSize.Y / 2
    local col = Color3.new(1,1,1)
    local size = 2

    if ctype == "Dot" then
        local c = Drawing.new("Circle")
        c.Position = Vector2.new(cx, cy)
        c.Radius   = 3
        c.Color    = col
        c.Filled   = true
        c.Visible  = true
        table.insert(crosshairObjs, c)

    elseif ctype == "Cross" then
        local lines = {
            {Vector2.new(cx-10,cy), Vector2.new(cx+10,cy)},
            {Vector2.new(cx,cy-10), Vector2.new(cx,cy+10)},
        }
        for _, l in ipairs(lines) do
            local d = Drawing.new("Line")
            d.From = l[1]; d.To = l[2]
            d.Color = col; d.Thickness = size; d.Visible = true
            table.insert(crosshairObjs, d)
        end

    elseif ctype == "Circle" then
        local c = Drawing.new("Circle")
        c.Position = Vector2.new(cx, cy)
        c.Radius   = 12; c.Color = col; c.Filled = false
        c.Thickness = size; c.Visible = true
        table.insert(crosshairObjs, c)

    elseif ctype == "Diamond" then
        local pts = {
            Vector2.new(cx,cy-12), Vector2.new(cx+12,cy),
            Vector2.new(cx,cy+12), Vector2.new(cx-12,cy),
            Vector2.new(cx,cy-12),
        }
        for i = 1, #pts-1 do
            local d = Drawing.new("Line")
            d.From = pts[i]; d.To = pts[i+1]
            d.Color = col; d.Thickness = size; d.Visible = true
            table.insert(crosshairObjs, d)
        end

    elseif ctype == "T-Shape" then
        local lines = {
            {Vector2.new(cx-10,cy), Vector2.new(cx+10,cy)},
            {Vector2.new(cx,cy),    Vector2.new(cx,cy+10)},
        }
        for _, l in ipairs(lines) do
            local d = Drawing.new("Line")
            d.From = l[1]; d.To = l[2]
            d.Color = col; d.Thickness = size; d.Visible = true
            table.insert(crosshairObjs, d)
        end

    elseif ctype == "Sniper" then
        local lines = {
            {Vector2.new(cx-20,cy), Vector2.new(cx-5,cy)},
            {Vector2.new(cx+5,cy),  Vector2.new(cx+20,cy)},
            {Vector2.new(cx,cy-20), Vector2.new(cx,cy-5)},
            {Vector2.new(cx,cy+5),  Vector2.new(cx,cy+20)},
        }
        for _, l in ipairs(lines) do
            local d = Drawing.new("Line")
            d.From = l[1]; d.To = l[2]
            d.Color = col; d.Thickness = size; d.Visible = true
            table.insert(crosshairObjs, d)
        end
        -- Center dot
        local c = Drawing.new("Circle")
        c.Position = Vector2.new(cx,cy)
        c.Radius = 2; c.Color = col; c.Filled = true; c.Visible = true
        table.insert(crosshairObjs, c)
    end

    S.crosshairType = ctype
end

-- Redraw crosshair on viewport resize
Cam:GetPropertyChangedSignal("ViewportSize"):Connect(function()
    if S.crosshairType ~= "None" then
        drawCrosshair(S.crosshairType)
    end
end)

-- Serialize current state
local function serializeState()
    return {
        walkSpeed     = S.walkSpeed,
        parryRange    = S.parryRange,
        faceAngle     = S.faceKillerAngle,
        parryDelay    = S.parryDelay,
        aimRadius     = S.aimRadius,
        aimSmooth     = S.aimSmooth,
        silentTarget  = S.silentAimTarget,
        fov           = S.fov,
        crosshair     = S.crosshairType,
        noTexture     = S.noTexture,
        fullBright    = S.fullBright,
        noFog         = S.noFog,
    }
end

local function applyState(data)
    if not data then return end
    if data.walkSpeed   then setSpeed(data.walkSpeed) end
    if data.parryRange  then S.parryRange=data.parryRange end
    if data.faceAngle   then S.faceKillerAngle=data.faceAngle end
    if data.parryDelay  then S.parryDelay=data.parryDelay end
    if data.aimRadius   then S.aimRadius=data.aimRadius end
    if data.aimSmooth   then S.aimSmooth=data.aimSmooth end
    if data.silentTarget then S.silentAimTarget=data.silentTarget end
    if data.fov         then Cam.FieldOfView=data.fov; S.fov=data.fov end
    if data.crosshair   then drawCrosshair(data.crosshair) end
    if data.noTexture   then setNoTexture(data.noTexture) end
    if data.fullBright  then setFullBright(data.fullBright) end
    if data.noFog       then setNoFog(data.noFog) end
end
