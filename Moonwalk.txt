local function worldToVP(pos)

-- ════════════════════════════════════════════════════════════════
--  FLY
-- ════════════════════════════════════════════════════════════════
local function setFly(v)
    S.Fly = v
    if S.FlyConn then
        S.FlyConn:Disconnect()
        S.FlyConn = nil
    end

    local hum = getHum()
    local hrp = getHRP()
    if not hum or not hrp then return end

    if v then
        safe(function() hum.PlatformStand = true end)

        local bg = Instance.new("BodyGyro")
        bg.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
        bg.P = 1e4
        bg.D = 1e3
        bg.Parent = hrp

        local bv = Instance.new("BodyVelocity")
        bv.MaxForce = Vector3.new(1e9, 1e9, 1e9)
        bv.Velocity = V3ZERO
        bv.Parent = hrp

        S.FlyConn = RunService.Heartbeat:Connect(function()
            local hrp2 = getHRP()
            if not hrp2 then return end
            if not bg or not bg.Parent then return end
            if not bv or not bv.Parent then return end

            local spd = S.FlySpeed
            local dir = V3ZERO
            local cf  = Camera.CFrame

            if UserInputService:IsKeyDown(Enum.KeyCode.W) then
                dir = dir + cf.LookVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then
                dir = dir - cf.LookVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then
                dir = dir - cf.RightVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then
                dir = dir + cf.RightVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
                dir = dir + Vector3.new(0, 1, 0)
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl)
            or UserInputService:IsKeyDown(Enum.KeyCode.ButtonL2) then
                dir = dir - Vector3.new(0, 1, 0)
            end

            safe(function()
                if dir.Magnitude > 0 then
                    bv.Velocity = dir.Unit * spd
                else
                    bv.Velocity = V3ZERO
                end
                bg.CFrame = Camera.CFrame
            end)
        end)
    else
        if hrp then
            for _, c in ipairs(hrp:GetChildren()) do
                if c:IsA("BodyGyro") or c:IsA("BodyVelocity") then
                    c:Destroy()
                end
            end
        end
        safe(function() if hum then hum.PlatformStand = false end end)
    end
end

-- ════════════════════════════════════════════════════════════════
--  TARGET DETECTION (Silent Aim / ESP)
-- ════════════════════════════════════════════════════════════════
local function isCharKiller(char)
    if not char then return false end
    return char:FindFirstChild("KillerConfig") ~= nil
        or char:FindFirstChild("KillerMorph") ~= nil
end

local function getBestTarget(targetType)
    local best     = nil
    local bestDist = S.AimRadius
    local center   = screenCenter()

    for _, p in ipairs(Players:GetPlayers()) do
        if p == LP then goto nextPlayer end
        local char = p.Character
        if not char then goto nextPlayer end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hrp or not hum or hum.Health <= 0 then goto nextPlayer end

        local killerChar = isCharKiller(char)
        local pass = false
        if targetType == "Killer"   and killerChar then pass = true end
        if targetType == "Survivor" and not killerChar then pass = true end
        if targetType == "All" then pass = true end
        if not pass then goto nextPlayer end

        do
            local vpPos, inView = worldToVP(hrp.Position)
            if not inView then goto nextPlayer end
            local dist = (vpPos - center).Magnitude
            if dist < bestDist then
                bestDist = dist
                best = hrp
            end
        end

        ::nextPlayer::
    end
    return best
end

-- ════════════════════════════════════════════════════════════════
--  AUTO GENERATOR
-- ════════════════════════════════════════════════════════════════
local function findNearestGen()
    local hrp = getHRP()
    if not hrp then return nil end
    local best, bestD = nil, math.huge

    for _, obj in ipairs(WS:GetDescendants()) do
        local n = obj.Name
        if n == "Generator" or n == "Gen" or n == "GeneratorPoint" then
            if obj:IsA("BasePart") or obj:IsA("Model") then
                local pos
                safe(function()
                    pos = obj:IsA("Model") and obj:GetModelCFrame().Position or obj.Position
                end)
                if pos then
                    local d = (pos - hrp.Position).Magnitude
                    if d < bestD then bestD = d best = obj end
                end
            end
        end
    end
    return best
end

local function fireRepairEvent()
    safe(function()
        local rs = ReplicatedStorage
        local ev = rs:FindFirstChild("RepairEvent", true)
            or rs:FindFirstChild("GeneratorRepair", true)
            or rs:FindFirstChild("Repair", true)
        if ev then
            if ev:IsA("RemoteEvent") then ev:FireServer()
            elseif ev:IsA("BindableEvent") then ev:Fire() end
        end
        local sk = rs:FindFirstChild("SkillCheckResultEvent", true)
            or rs:FindFirstChild("SkillcheckValidated", true)
        if sk then
            if sk:IsA("RemoteEvent") then sk:FireServer(true)
            elseif sk:IsA("BindableEvent") then sk:Fire(true) end
        end
    end)
end

local function setAutoGen(v)
    S.AutoGen = v
    if S.GenConn then S.GenConn:Disconnect() S.GenConn = nil end
    if not v then return end
    S.GenConn = RunService.Heartbeat:Connect(function()
        if not S.AutoGen then return end
        local gen = findNearestGen()
        if not gen then return end
        local hrp = getHRP()
        if not hrp then return end
        local pos
        safe(function()
            pos = gen:IsA("Model") and gen:GetModelCFrame().Position or gen.Position
        end)
        if pos and (pos - hrp.Position).Magnitude < 6 then
            fireRepairEvent()
        end
    end)
end

-- ════════════════════════════════════════════════════════════════

-- ════════════════════════════════════════════════════════════════
local function doParry()
    safe(function()
        local rs = ReplicatedStorage
        local ev = rs:FindFirstChild("ParryEvent", true)
            or rs:FindFirstChild("AttackEvent", true)
            or rs:FindFirstChild("BasicAttack", true)
        if ev then
            if ev:IsA("RemoteEvent") then ev:FireServer("Parry")
            elseif ev:IsA("BindableEvent") then ev:Fire("Parry") end
        end
    end)
end

local function killerFacing(killerHRP)
    local myHRP = getHRP()
    if not myHRP or not killerHRP then return true end
    local toSurv = (myHRP.Position - killerHRP.Position)
    if toSurv.Magnitude < 0.001 then return true end
    toSurv = toSurv.Unit
    local look = killerHRP.CFrame.LookVector
    local dot   = math.clamp(look:Dot(toSurv), -1, 1)
    local angle = math.deg(math.acos(dot))
    return angle <= (S.FaceKillerAngle / 2)
end

local function setAutoParry(v)
    S.AutoParry = v
    if S.ParryConn then S.ParryConn:Disconnect() S.ParryConn = nil end
    if not v then return end
    S.ParryConn = RunService.Heartbeat:Connect(function()
        if not S.AutoParry then return end
        local myHRP = getHRP()
        if not myHRP then return end
        local kHRP = findKillerHRP()
        if not kHRP then return end
        local dist = (kHRP.Position - myHRP.Position).Magnitude
        if dist <= S.ParryRange then
            if not killerFacing(kHRP) then return end
            task.wait(S.ParryDelay)
            doParry()
        end
    end)
end

--  KILLER: SNAP LINE
-- ════════════════════════════════════════════════════════════════
local _snapLineConn = nil
local function buildSnapLine()
    if _snapLineConn then _snapLineConn:Disconnect() _snapLineConn = nil end
    local pgui = LP:FindFirstChild("PlayerGui")
    if not pgui then return end
    local old = pgui:FindFirstChild("RYZ_SnapLine")
    if old then old:Destroy() end
    if not S.SnapLine then return end

    local sg = Instance.new("ScreenGui")
    sg.Name = "RYZ_SnapLine"
    sg.ResetOnSpawn = false
    sg.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    sg.DisplayOrder = 9998
    sg.IgnoreGuiInset = true
    sg.Parent = pgui

    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, 0, 1, 0)
    frame.BackgroundTransparency = 1
    frame.Parent = sg

    _snapLineConn = RunService.RenderStepped:Connect(function()
        if not S.SnapLine or not frame.Parent then
            frame:ClearAllChildren()
            return
        end
        frame:ClearAllChildren()

        local target = getBestTarget("Survivor")
        if not target then return end
        local myHRP = getHRP()
        if not myHRP then return end

        local vpA, inA = worldToVP(myHRP.Position)
        local vpB, inB = worldToVP(target.Position)
        if not inA or not inB then return end

        local dist  = (vpB - vpA).Magnitude
        local angle = math.atan2(vpB.Y - vpA.Y, vpB.X - vpA.X)
        local mid   = (vpA + vpB) / 2

        local line = Instance.new("Frame")
        line.Size = fromOffset(dist, 2)
        line.Position = fromOffset(mid.X - dist / 2, mid.Y - 1)
        line.AnchorPoint = Vector2.new(0.5, 0.5)
        line.BackgroundColor3 = Color3.fromRGB(255, 60, 60)
        line.BackgroundTransparency = 0.3
        line.BorderSizePixel = 0
        line.Rotation = math.deg(angle)
        line.ZIndex = 5
        local uc = Instance.new("UICorner")
        uc.CornerRadius = UDim.new(1, 0)
        uc.Parent = line
        line.Parent = frame
    end)
end

local ESPData = {}

local function removeESP(model)
    local data = ESPData[model]
    if not data then return end
    safe(function()
        if data.hl and data.hl.Parent then data.hl:Destroy() end
    end)
    safe(function()
        if data.bb and data.bb.Parent then data.bb:Destroy() end
    end)
    ESPData[model] = nil
end

local function makeHL(adornee, outline, fill, fillTrans)
    local hl = Instance.new("SelectionBox")
    hl.Adornee = adornee
    hl.Color3 = outline
    hl.SurfaceColor3 = fill
    hl.SurfaceTransparency = fillTrans or 0.75
    hl.LineThickness = 0.04
    hl.Parent = WS
    return hl
end

local function makeBB(part, color)
    if not part or not part.Parent then return nil, nil end
    local bg = Instance.new("BillboardGui")
    bg.Adornee = part
    bg.AlwaysOnTop = true
    bg.Size = fromOffset(200, 56)
    bg.StudsOffsetWorldSpace = Vector3.new(0, 3.5, 0)
    bg.ResetOnSpawn = false
    bg.Parent = part

    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(1, 0, 1, 0)
    lbl.BackgroundTransparency = 1
    lbl.Text = ""
    lbl.TextColor3 = color or Color3.fromRGB(255, 255, 255)
    lbl.TextSize = 12
    lbl.Font = Enum.Font.GothamBold
    lbl.TextStrokeTransparency = 0
    lbl.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    lbl.Parent = bg
    return bg, lbl
end

local function refreshSurvivorESP()
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LP then goto nextSE end
        local char = p.Character
        if not char then goto nextSE end
        if isCharKiller(char) then goto nextSE end

        if S.ESPSurvivor then
            if not ESPData[char] then
                local hrp = char:FindFirstChild("HumanoidRootPart")
                local head = char:FindFirstChild("Head")
                local col = Color3.fromRGB(0, 180, 255)
                local hl = makeHL(char, col, col, 0.8)
                local bb, lbl = makeBB(hrp or head, col)
                ESPData[char] = {hl=hl, bb=bb, lbl=lbl, type="Survivor"}
            end
        else
            if ESPData[char] and ESPData[char].type == "Survivor" then
                removeESP(char)
            end
        end
        ::nextSE::
    end
end

local function refreshKillerESP()
    for _, p in ipairs(Players:GetPlayers()) do

local function refreshGenESP()
    if not S.ESPGenerator then
        -- copy keys first to avoid mutation during iteration
        local toRemove = {}
        for m, d in pairs(ESPData) do
            if d.type == "Generator" then toRemove[#toRemove+1] = m end
        end
        for _, m in ipairs(toRemove) do removeESP(m) end
        return
    end
    for _, obj in ipairs(WS:GetDescendants()) do
        if (obj.Name == "Generator" or obj.Name == "Gen") and obj:IsA("Model") then
            if not ESPData[obj] then
                local col = Color3.fromRGB(255, 200, 0)
                local pp = obj.PrimaryPart or obj:FindFirstChildOfClass("BasePart")
                if pp then
                    local hl = makeHL(obj, col, col, 0.75)
                    local bb, lbl = makeBB(pp, col)
                    ESPData[obj] = {hl=hl, bb=bb, lbl=lbl, type="Generator"}
                end
            end
        end
    end
end

local function refreshHookESP()
    if not S.ESPHook then
        local toRemove = {}
        for m, d in pairs(ESPData) do
            if d.type == "Hook" then toRemove[#toRemove+1] = m end

        for _, m in ipairs(toRemove) do removeESP(m) end

    for _, obj in ipairs(WS:GetDescendants()) do

        end
    end
end

local CROSSHAIRS = {
    Dot     = {shape="dot",    size=6,  color=Color3.fromRGB(255,255,255)},
    Cross   = {shape="cross",  size=18, color=Color3.fromRGB(255,255,255)},
    Circle  = {shape="circle", size=20, color=Color3.fromRGB(255,255,255)},
    ["T-Shape"] = {shape="tshape", size=18, color=Color3.fromRGB(255,80,80)},
    Plus    = {shape="plus",   size=18, color=Color3.fromRGB(0,220,100)},
    None    = {shape="none",   size=0,  color=Color3.fromRGB(0,0,0)},
}

local function buildCrosshair(style)
    local pgui = LP:FindFirstChild("PlayerGui")
    if not pgui then return end
    local old = pgui:FindFirstChild("RYZ_Crosshair")
    if old then old:Destroy() end
    S.CrosshairGui = nil
    if not style or style == "None" then return end

    local cfg = CROSSHAIRS[style]
    if not cfg or cfg.shape == "none" then return end

    local sg = Instance.new("ScreenGui")
    sg.Name = "RYZ_Crosshair"
    sg.ResetOnSpawn = false
    sg.DisplayOrder = 9990
    sg.IgnoreGuiInset = true
    sg.Parent = pgui
    S.CrosshairGui = sg

    local vp = Camera.ViewportSize
    local cx, cy = vp.X / 2, vp.Y / 2
    local s   = cfg.size
    local col = cfg.color

    local function rect(x, y, w, h)
        local f = Instance.new("Frame")
        f.Position = fromOffset(x, y)
        f.Size = fromOffset(w, h)
        f.BackgroundColor3 = col
        f.BorderSizePixel = 0
        f.ZIndex = 10
        f.Parent = sg
        return f
    end

    if cfg.shape == "dot" then
        local r = rect(cx - s/2, cy - s/2, s, s)
        local uc = Instance.new("UICorner")
        uc.CornerRadius = UDim.new(1, 0)
        uc.Parent = r

    elseif cfg.shape == "cross" or cfg.shape == "plus" then
        rect(cx - s/2, cy - 1, s, 2)
        rect(cx - 1, cy - s/2, 2, s)

    elseif cfg.shape == "circle" then
        local f = Instance.new("Frame")
        f.Position = fromOffset(cx - s/2, cy - s/2)
        f.Size = fromOffset(s, s)
        f.BackgroundTransparency = 1
        f.BorderSizePixel = 0
        f.ZIndex = 10
        f.Parent = sg
        local ui = Instance.new("UICorner")
        ui.CornerRadius = UDim.new(1, 0)
        ui.Parent = f
        local st = Instance.new("UIStroke")
        st.Color = col
        st.Thickness = 1.5
        st.Parent = f

    elseif cfg.shape == "tshape" then
        rect(cx - s/2, cy, s, 2)
        rect(cx - 1, cy - s/2, 2, s/2)
    end
end

local function setNoTexture(v)
    S.NoTexture = v
    safe(function()
        if v then
            for _, obj in ipairs(WS:GetDescendants()) do
                if obj:IsA("Texture") or obj:IsA("Decal") then
                    if not S.OrigTextures[obj] then
                        S.OrigTextures[obj] = obj.Transparency
                    end
                    obj.Transparency = 1
                end
            end
        else
            for obj, orig in pairs(S.OrigTextures) do
                safe(function()
                    if obj and obj.Parent then obj.Transparency = orig end
                end)
            end
            S.OrigTextures = {}
        end
    end)
end

local CONFIG_FOLDER = "RYZ_VD_Configs"

local function ensureConfigFolder()
    if not HAS_FILE_API then return end
    if not _isfolder(CONFIG_FOLDER) then _makefolder(CONFIG_FOLDER) end
end

local function getConfigList()
    if not HAS_FILE_API then return {"(File API N/A)"} end
    ensureConfigFolder()
    local list = {}
    for _, f in ipairs(_listfiles(CONFIG_FOLDER)) do
        local n = tostring(f):match("([^/\\]+)$") or tostring(f)
        if n:sub(-5) == ".json" then
            list[#list + 1] = n:sub(1, -6)
        end
    end
    return #list > 0 and list or {"(No Configs)"}
end

local function serializeConfig()
    return HttpService:JSONEncode({
        NoClip=S.NoClip, Fly=S.Fly, FlySpeed=S.FlySpeed,
        SpeedHack=S.SpeedHack, SpeedValue=S.SpeedValue,
        InfJump=S.InfJump, AntiAFK=S.AntiAFK,
        AutoParry=S.AutoParry, ParryRange=S.ParryRange,
        FaceKillerAngle=S.FaceKillerAngle, ParryDelay=S.ParryDelay,
        SilentAim=S.SilentAim, AimRadius=S.AimRadius,
        AimSmooth=S.AimSmooth, SilentLaser=S.SilentLaser,
        AimTarget=S.AimTarget, AutoGen=S.AutoGen,
        KillerSilentAim=S.KillerSilentAim, SpearGravity=S.SpearGravity,
        SnapLine=S.SnapLine, AimPredict=S.AimPredict,
        AntiStun=S.AntiStun, AntiFlashlight=S.AntiFlashlight,
        ESPSurvivor=S.ESPSurvivor, ESPKiller=S.ESPKiller,
        ESPGenerator=S.ESPGenerator, ESPHook=S.ESPHook,
        ESPPallet=S.ESPPallet, KillerNotif=S.KillerNotif,
        CamFOV=S.CamFOV, Crosshair=S.Crosshair,
        NoTexture=S.NoTexture, FullBright=S.FullBright, NoFog=S.NoFog,
    })
end

local function saveConfig(name)
    if not HAS_FILE_API then return false, "File API tidak tersedia" end
    if not name or name == "" then return false, "Nama kosong" end
    ensureConfigFolder()
    local ok, err = _writefile(CONFIG_FOLDER .. "/" .. name .. ".json", serializeConfig())
    return ok, err
end

local function loadConfig(name)
    if not HAS_FILE_API then return false end
    local raw = _readfile(CONFIG_FOLDER .. "/" .. name .. ".json")
    if not raw then return false end
    local ok, data = pcall(function() return HttpService:JSONDecode(raw) end)
    if not ok or type(data) ~= "table" then return false end
    for k, v in pairs(data) do
        if S[k] ~= nil then S[k] = v end
    end
    -- re-apply
    setNoClip(S.NoClip)
    if S.Fly then setFly(false) task.wait(0.1) setFly(true)
    else setFly(false) end
    applySpeed()
    setAutoGen(S.AutoGen)
    setAutoParry(S.AutoParry)
    setSpearGravity(S.SpearGravity)
    setAntiStun(S.AntiStun)
    setAntiFlashlight(S.AntiFlashlight)
    buildSnapLine()
    setFOV(S.CamFOV)
    buildCrosshair(S.Crosshair)
    setNoTexture(S.NoTexture)
    setFullBright(S.FullBright)
    setNoFog(S.NoFog)
    return true
end

local function deleteConfig(name)
    if not HAS_FILE_API then return end
    _delfile(CONFIG_FOLDER .. "/" .. name .. ".json")
end

local TCfg = Window:Tab({ Title = "Config", Icon = "folder" })

local configName = ""
local selConfig  = ""
local configDD

local function refreshConfigDD()
    local list = getConfigList()
    selConfig = list[1] or ""
    safe(function() configDD:Refresh(list) end)
end

configDD = TCfg:Dropdown({
    Title  = "Config List", Icon = "list",
    Values = getConfigList(),
    Value  = getConfigList()[1],
    Callback = function(v) selConfig = v end,
})
TCfg:Input({
    Title       = "Nama Config", Icon = "file-text",
    Placeholder = "contoh: MySetting",
    Callback    = function(v) configName = v end,
})
TCfg:Button({
    Title = "Save Config", Icon = "save", Color = Color3.fromHex("#22c55e"),
    Callback = function()
        local name = configName ~= "" and configName or ("config_" .. os.time())
        local ok, err = saveConfig(name)
        if ok then
            notify("Config", "Tersimpan: " .. name, "save", 3)
            refreshConfigDD()
        else
            notify("Config", "Gagal simpan: " .. tostring(err), "alert-circle", 3)
        end
    end,
})
TCfg:Button({
    Title = "Load Config", Icon = "folder-open", Color = Color3.fromHex("#3b82f6"),
    Callback = function()
        if selConfig == "" or selConfig == "(No Configs)" or selConfig == "(File API N/A)" then
            notify("Config", "Pilih config dulu!", "alert-circle", 2) return
        end
        if loadConfig(selConfig) then
            notify("Config", "Loaded: " .. selConfig, "folder-open", 3)
        else
            notify("Config", "Gagal load!", "alert-circle", 3)
        end
    end,
})
TCfg:Button({
    Title = "Delete Config", Icon = "trash-2", Color = Color3.fromHex("#ef4444"),
    Callback = function()
        if selConfig == "" or selConfig == "(No Configs)" or selConfig == "(File API N/A)" then
            notify("Config", "Pilih config dulu!", "alert-circle", 2) return
        end
        deleteConfig(selConfig)
        notify("Config", "Dihapus: " .. selConfig, "trash-2", 3)
        refreshConfigDD()
    end,
})
TCfg:Button({
    Title = "Refresh List", Icon = "refresh-cw",
    Callback = function() refreshConfigDD() end,
})
TCfg:Divider()
